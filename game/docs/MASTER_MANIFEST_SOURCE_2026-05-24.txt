```javascript
const SVRCharityModule = {
    activeCampaigns: [],
    
    init() {
        console.log("SVR Charity Module Initialized.");

    },

    /**
     * Dynamically fetches real-time fundraising targets and progress metrics

    triggerWinnerDisplay(winnerUsername, handName, communityCards, winningCards) {

        // Construct the enterprise telemetry message
        const displayMessage = `POT WON BY: ${winnerUsername}\nHAND: ${handName}\nCARDS: ${winningCards.join(', ')}`;
        
        window.dispatchEvent(new CustomEvent('svr_show_banner', { 
            detail: { 
                message: displayMessage,
                duration: 10000 // Strict 10-second requirement
            } 
        }));

        // Reset highlights automatically after the window expires
        setTimeout(() => {
            window.dispatchEvent(new CustomEvent('svr_highlight_player', { detail: { username: winnerUsername, highlight: false } }));

    }
```

### 2. Private Community Module (`js/scarlett1/mod_private.js`)

Manages tester logins, synchronization of holistic **Reiki points** or driving range milestones, and handles **automated seating** (when a user steps into the play zone, they sit and get cards instantly at a **6-seat max table**).

```javascript
/**
 * SVR Poker - Private Community & Verification Module
 * Filename: js/scarlett1/mod_private.js
 * Version: 5.0
 */

const SVRPrivateModule = {
    sessionToken: null,
    playerProfile: null,

    init() {
        console.log("SVR Private Community Module Initialized.");
        this.checkStoredSession();
    },

    checkStoredSession() {
        const token = localStorage.getItem('svr_auth_token');
        if (token) {
            this.verifySession(token);
        }
    },

    async verifySession(token) {
        try {
            const response = await fetch('/api/auth/verify', {
                headers: { 'Authorization': `Bearer ${token}` }
            });
            if (response.ok) {
                this.sessionToken = token;
                this.playerProfile = await response.json();
                window.dispatchEvent(new CustomEvent('svr_user_authenticated', { detail: this.playerProfile }));
            } else {
                this.logout();
            }
        } catch (error) {
            console.error("Session authorization failure:", error);
        }
    },

    /**
     * Core Requirement: Automated Seating Logic
     * Triggers whenever positional coordinate thresholds register a seating event.
     */
    evaluateSeatingProximity(playerPosition) {
        if (!this.playerProfile) return;

        const playZoneCoordinates = { x: 0, z: 0, radius: 2.5 }; // Target "Play Game" interactive boundary
        const dx = playerPosition.x - playZoneCoordinates.x;
        const dz = playerPosition.z - playZoneCoordinates.z;
        const distance = Math.sqrt(dx * dx + dz * dz);

        if (distance <= playZoneCoordinates.radius) {
            this.executeAutoSeat();
        }
    },

    executeAutoSeat() {
        window.dispatchEvent(new CustomEvent('svr_request_seat', {
            detail: {
                userId: this.playerProfile.user_id,
                username: this.playerProfile.username,
                chips: this.playerProfile.chips_balance,
                maxSeats: 6 // Absolute configuration cap for the core poker pit layout
            }
        }));
    },

    /**
     * Synchronizes progression values from immersive experiences (Reiki rooms / PGA golf arrays)
     */
    async updateUserPoints(pointType, numericalValue) {
        if (!this.sessionToken) return;

        try {
            const response = await fetch('/api/user/progression', {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                    'Authorization': `Bearer ${this.sessionToken}`
                },
                body: JSON.stringify({ type: pointType, increment: numericalValue })
            });
            if (response.ok) {
                const updatedData = await response.json();
                this.playerProfile.reiki_points = updatedData.reiki_points;
            }
        } catch (error) {
            console.error("Failed to commit progression data:", error);
        }
    },

    logout() {
        localStorage.removeItem('svr_auth_token');
        this.sessionToken = null;
        this.playerProfile = null;
        window.location.reload();
    }
};

SVRPrivateModule.init();
```

### 3. Automated Sponsorship Module (`js/scarlett1/mod_sponsor.js`)

Queries your database for live campaigns, handles sponsor ad packets, and injects logo textures directly onto the table felts or 3D billboards.

```javascript
/**
 * SVR Poker - Automated Sponsorship Engine
 * Filename: js/scarlett1/mod_sponsor.js
 * Version: 5.0
 */

const SVRSponsorModule = {
    activeTextures: {},

    init() {
        console.log("SVR Automated Sponsorship System Initialized.");
        this.loadSponsorAssets();
    },

    /**
     * Queries the database population system for active, compliance-verified assets
     */
    async loadSponsorAssets() {
        try {
            const response = await fetch('/api/sponsors/active-campaigns');
            if (response.ok) {
                const campaigns = await response.json();
                this.processCampaignData(campaigns);
            }
        } catch (error) {
            console.error("Failed loading automated partner assets:", error);
        }
    },

    /**
     * Maps database entries directly into structured web paths
     */
    processCampaignData(campaigns) {
        campaigns.forEach(ad => {
            if (ad.status === 'active' && ad.logo_url) {
                this.activeTextures[ad.ad_type] = ad.logo_url;
            }
        });

        this.injectSponsorTextures();
    },

    /**
     * Safely announces new texture states to world.js without disrupting active render operations
     */
    injectSponsorTextures() {
        if (Object.keys(this.activeTextures).length === 0) return;

        window.dispatchEvent(new CustomEvent('svr_inject_sponsor_materials', {
            detail: { textures: this.activeTextures }
        }));
    },

    /**
     * Verification routine checking programmatic compliance of active agreements
     */
    verifyCampaignCompliance(campaignId) {
        return fetch(`/api/sponsors/compliance/${campaignId}`)
            .then(res => res.json())
            .then(data => data.isCompliant)
            .catch(() => false);
    }
};

SVRSponsorModule.init();
```

```javascript
        } finally {
            window.dispatchEvent(new CustomEvent('svr_commerce_loading', { detail: { active: false } }));
        }
        return false;
    }
};
```

## 2. Cosmic Skybox: Dynamic Moon & Mars Orbit Engine

To fix the blurriness of your space elements, we switch from standard flat skybox images to **procedural mipmapped textures** wrapped around discrete high-density spheres. This math-driven path forces the Moon to swing close to your viewpoint (revealing massive geometric depth) before looping far behind Mars, completely separate from your steady Chicago skyline buildings.

### Add this physics block to your permanent rendering engine (`world.js`):

```javascript
/**
 * SVR Cosmic Orbit & Scale Engine
 * Coordinates planetary rotation matrices independent of background city meshes.
 */
const SVRCosmicEngine = {
    marsSphere: null,
    moonSphere: null,
    orbitClock: 0,

    initCelestialBodies(scene) {
        // High-precision texture loading to eliminate blurriness
        const textureLoader = new THREE.TextureLoader();
        
        // Mars Setup
        const marsGeo = new THREE.SphereGeometry(40, 32, 32);
        const marsMat = new THREE.MeshStandardMaterial({
            map: textureLoader.load('assets/textures/space/mars_highres.png'),
            roughness: 0.8,
            metalness: 0.1
        });
        this.marsSphere = new THREE.Mesh(marsGeo, marsMat);
        // Position Mars far out in the background field
        this.marsSphere.position.set(0, 120, -400); 
        scene.add(this.marsSphere);

        // Moon Setup
        const moonGeo = new THREE.SphereGeometry(12, 64, 64); // Higher subdivision prevents edge blurring
        const moonMat = new THREE.MeshStandardMaterial({
            map: textureLoader.load('assets/textures/space/moon_highres.png'),
            roughness: 0.9,
            bumpScale: 1.5,
            metalness: 0.0
        });
        this.moonSphere = new THREE.Mesh(moonGeo, moonMat);
        scene.add(this.moonSphere);
    },

    updateOrbits(deltaTime) {
        if (!this.marsSphere || !this.moonSphere) return;

        // Constant axial rotation
        this.marsSphere.rotation.y += deltaTime * 0.02;
        this.moonSphere.rotation.y += deltaTime * 0.05;

        // Complex orbital math: Varies distance (Z) and size perception over time
        this.orbitClock += deltaTime * 0.15; // Controls path speed

        // Elliptical Math Coordinates relative to Mars center position
        const radiusX = 140; 
        const radiusZ = 220; 

        const localX = Math.sin(this.orbitClock) * radiusX;
        const localZ = Math.cos(this.orbitClock) * radiusZ;

        // Apply spatial translation
        this.moonSphere.position.x = this.marsSphere.position.x + localX;
        this.moonSphere.position.y = this.marsSphere.position.y + (Math.sin(this.orbitClock) * 30);
        this.moonSphere.position.z = this.marsSphere.position.z + localZ;

        // Dynamic Scale Interpolation: Simulates the Moon getting massive when passing close to the player
        // As it approaches the foreground (closer Z), it scales up programmatically
        const distanceToPlayer = this.moonSphere.position.z;
        const scaleFactor = THREE.MathUtils.mapLinear(distanceToPlayer, -620, -180, 0.6, 2.8);
        this.moonSphere.scale.setScalar(scaleFactor);
    }
};
```

```javascript
function checkHandOrientation(leftHandMesh, camera) {
    // Get the upward direction vector of the hand plane
    const localUp = new THREE.Vector3(0, 1, 0);
    localUp.applyQuaternion(leftHandMesh.quaternion);

    const toCamera = new THREE.Vector3().subVectors(camera.position, leftHandMesh.position).normalize();

        detail: { facingUp: isPalmFacingFace } 
    }));
}
```

## 2. The Table Routing Module (`js/scarlett1/mod_router.js`)

This module intercepts players during the database population sequence and instantly pushes them to the exact 6-seat table mesh matching their requested game type inside the Scorpion Room.

```javascript
/**
 * SVR Poker - Smart Table Router & Population Engine
 * Filename: js/scarlett1/mod_router.js
 * Version: 1.0 (Scorpion Room Optimized)
 */

const SVRTableRouter = {
    roomTables: [],

    init() {
        console.log("SVR Smart Table Router Initialized.");
        
        // Listen for new database population events
        window.addEventListener('svr_database_population', (event) => {
            const incomingPlayers = event.detail.players;
            this.routePlayersToCorrectTables(incomingPlayers);
        });
    },

    /**
     * Queries active database states for the Scorpion Room configurations
     */
    async fetchScorpionRoomAvailability() {
        try {
            const response = await fetch('/api/rooms/scorpion/tables');
            if (response.ok) {
                this.roomTables = await response.json();
            }
        } catch (error) {
            console.error("Failed to populate active Scorpion Room matrix:", error);
        }
    },

    /**
     * Core Logic: Matches players to specialized tables rapidly
     */
    routePlayersToCorrectTables(playerQueue) {
        if (this.roomTables.length === 0) {
            this.fetchScorpionRoomAvailability();
        }

        playerQueue.forEach(player => {
            // Find an open table matching the player's target game style (e.g., Bounty, Classic)
            const optimalTable = this.roomTables.find(table => 
                table.game_type === player.preferred_game_type && 
                table.current_occupancy < table.seat_count &&
                table.room_zone === 'scorpion_room'
            );

            if (optimalTable) {
                this.assignPlayerToTableMesh(player, optimalTable);
            } else {
                console.warn(`No available variant table found for player: ${player.username}`);
            }
        });
    },

    /**
     * Injects the player directly into the designated 3D table structure
     */
    assignPlayerToTableMesh(player, table) {
        table.current_occupancy++;
        
        // Broadcast immediately to world.js to render the player at the physical table asset
        window.dispatchEvent(new CustomEvent('svr_render_seat_allocation', {
            detail: {
                userId: player.user_id,
                username: player.username,
                targetTableId: table.table_id,
                gameType: table.game_type,
                seatIndex: table.current_occupancy,
                maxSeats: 6 // Adhering strictly to your 6-seat max constraint
            }
        }));
    }
};

// Auto-initialize
SVRTableRouter.init();

```

## 3. Gesture-Driven "Peek & Squeeze" Card Shader

With a purely hands-only interface, picking up cards flat off a digital felt can feel unnatural. To match the premium look of the showroom table's deep divots, we can implement a vertex-displacement card shader model.

When a player's tracked index finger and thumb pinch the corner of their two down-cards and lift up, the card model does not move rigidly. Instead, the mesh **bends and flexes** dynamically—replicating the physical tension of checking hole cards in a real cardroom.

```
       [ VR HAND INTERACTION ]
          (Pinch Gesture)
                │
                ▼
       [ HOLE CARD GEOMETRY ]
     ┌──────────────────────┐ 🔼 (Vertex Displacement)
     │ ♠                    │ │  The mesh bends upward
     │   ┌──────────────┐   │ │  to reveal values while
     │   │  CARD BACK   │   │ │  the base stays flat
     └───┴──────────────┴───┘    on the table felt.

```

### Mechanical Implementation Framework:

* **The Math Matrix:** Your rendering engine (`world.js`) watches the distance between your tracked thumb bone and the table felt surface.
* **The Vertex Hook:** As that distance increases during a pinch state, a displacement modifier scales along the card's local Y-axis geometry, lifting only the top 30% of the mesh while the remaining base stays firmly pinned down. This completely eliminates clipping errors and tracking failures caused by lifting full cards into the air.

```javascript
                this.deactivateHologram();
            }
        });

    buildHolographicInterface(handAnchor) {
        this.holoMenuMesh = new THREE.Group();
        this.holoMenuMesh.position.set(0, 0.15, 0); // Floats Menu comfortably above the wrist tracking mesh
        this.holoMenuMesh.visible = false;

        // Visual Projector Base Ring
        const ringGeo = new THREE.TorusGeometry(0.08, 0.005, 8, 24);
        const ringMat = new THREE.MeshBasicMaterial({ color: 0x00ffcc, wireframe: true, transparent: true, opacity: 0.4 });
        const baseRing = new THREE.Mesh(ringGeo, ringMat);
        baseRing.rotation.x = Math.PI / 2;
        this.holoMenuMesh.add(baseRing);

        // Circular Layout Generation for Large Hit-Targets
        this.menuOptions.forEach((optionName, index) => {
            const angle = (index / this.menuOptions.length) * Math.PI * 2;
            const panelGeo = new THREE.BoxGeometry(0.05, 0.02, 0.005);
            const panelMat = new THREE.MeshBasicMaterial({ color: 0x2d1442, transparent: true, opacity: 0.8, side: THREE.DoubleSide });
            const panel = new THREE.Mesh(panelGeo, panelMat);
            
            panel.position.x = Math.cos(angle) * 0.06;
            panel.position.z = Math.sin(angle) * 0.06;
            panel.name = `btn_${optionName.toLowerCase()}`;
            
            this.holoMenuMesh.add(panel);
        });

        handAnchor.add(this.holoMenuMesh);
    },

    activateHologram() {
        this.isActive = true;
        this.holoMenuMesh.visible = true;
        window.dispatchEvent(new CustomEvent('svr_play_spatial_sfx', { detail: { type: 'ui_holo_up' } }));

        this.isActive = false;
        this.holoMenuMesh.visible = false;
```

```javascript
                        username: player.username,
                        targetTableId: table.table_id,
                        gameType: table.game_type,
                        seatIndex: table.current_occupancy,
```

```text
const SVRPrivateModule = {
    sessionToken: null,
    playerProfile: null,
    init() {
        console.log("SVR Private Community Module Initialized.");
        this.checkStoredSession();
    },
    checkStoredSession() {
        const token = localStorage.getItem('svr_auth_token');
        if (token) this.verifySession(token);
    },
    async verifySession(token) {
        try {
            const response = await fetch('/api/auth/verify', { headers: { 'Authorization': `Bearer ${token}` } });
            if (response.ok) {
                this.sessionToken = token;
                this.playerProfile = await response.json();
                window.dispatchEvent(new CustomEvent('svr_user_authenticated', { detail: this.playerProfile }));
            } else { this.logout(); }
        } catch (error) { console.error("Session authorization failure:", error); }
    },
    evaluateSeatingProximity(playerPosition) {
        if (!this.playerProfile) return;
        const playZoneCoordinates = { x: 0, z: 0, radius: 2.5 };
        const dx = playerPosition.x - playZoneCoordinates.x;
        const dz = playerPosition.z - playZoneCoordinates.z;
        if (Math.sqrt(dx * dx + dz * dz) <= playZoneCoordinates.radius) { this.executeAutoSeat(); }
    },
    executeAutoSeat() {
        window.dispatchEvent(new CustomEvent('svr_request_seat', {
            detail: { userId: this.playerProfile.user_id, username: this.playerProfile.username, chips: this.playerProfile.chips_balance, maxSeats: 6 }
        }));
    },
    async updateUserPoints(pointType, numericalValue) {
        if (!this.sessionToken) return;
        try {
            const response = await fetch('/api/user/progression', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${this.sessionToken}` },
                body: JSON.stringify({ type: pointType, increment: numericalValue })
            });
            if (response.ok) {
                const updatedData = await response.json();
                this.playerProfile.reiki_points = updatedData.reiki_points;
            }
        } catch (error) { console.error("Failed to commit progression data:", error); }
    },
    logout() {
        localStorage.removeItem('svr_auth_token');
        window.location.reload();
    }
};
```

```javascript
/**
 * SVR Poker - Multiplayer Avatar & Cosmetic Sync Engine
 * Filename: js/scarlett1/mod_avatar.js
 * Version: 1.0
 */

const SVRAvatarModule = {
    localAvatar: null,
    remoteAvatars: {}, // Holds active multiplayer session entities

    init() {
        console.log("SVR Avatar & Cosmetic Engine Initialized.");

        // Listen for successful profile loads to build the local tester avatar
        window.addEventListener('svr_user_authenticated', (event) => {
            this.createLocalAvatar(event.detail.user_id);
        });

        // Listen for storefront cosmetic changes to update looks instantly
        window.addEventListener('svr_commerce_success', (event) => {
            const purchase = event.detail; // Contains item_id, slot, and asset_path
            this.equipCosmeticItem(purchase.userId, purchase.slot, purchase.asset_path);
        });

        // Listen for incoming multiplayer network packet telemetries
        window.addEventListener('svr_network_player_update', (event) => {
            this.syncRemoteAvatar(event.detail);
        });
    },

    /**
     * Spawns a low-poly head and hands representation for network testing
     */
    createLocalAvatar(userId) {
        this.localAvatar = new THREE.Group();
        this.localAvatar.name = `avatar_${userId}`;

        // Low-poly head capsule template
        const headGeo = new THREE.ConeGeometry(0.12, 0.3, 6); // Faceted geometric look
        const headMat = new THREE.MeshStandardMaterial({ color: 0x2d1442, roughness: 0.5 });
        const headMesh = new THREE.Mesh(headGeo, headMat);
        headMesh.name = "head_mesh";
        this.localAvatar.add(headMesh);

        console.log("Local testing avatar generated successfully.");
    },

    /**
     * Spawns or updates other networked players sitting at your 6-seat table
     */
    syncRemoteAvatar(packet) {
        // packet: { userId, headPos, leftHandPos, rightHandPos, equippedItems }
        let avatar = this.remoteAvatars[packet.userId];

        if (!avatar) {
            // Instantly build a new low-poly avatar representation if they just joined the room
            avatar = new THREE.Group();
            
            const headGeo = new THREE.BoxGeometry(0.2, 0.2, 0.2);
            const headMat = new THREE.MeshStandardMaterial({ color: 0x1c1a21 });
            const head = new THREE.Mesh(headGeo, headMat);
            head.name = "head_mesh";
            avatar.add(head);

            // Left Hand Visual Target Placeholder
            const handGeo = new THREE.SphereGeometry(0.05, 6, 6);
            const handMat = new THREE.MeshStandardMaterial({ color: 0x00ffcc, wireframe: true });
            
            const leftHand = new THREE.Mesh(handGeo, handMat);
            leftHand.name = "left_hand_mesh";
            avatar.add(leftHand);

            const rightHand = leftHand.clone();
            rightHand.name = "right_hand_mesh";
            avatar.add(rightHand);

            window.dispatchEvent(new CustomEvent('svr_add_mesh_to_scene', { detail: { mesh: avatar } }));
            this.remoteAvatars[packet.userId] = avatar;
            
            // Render their default equipped items from the database
            if (packet.equippedItems) {
                packet.equippedItems.forEach(item => {
                    this.equipCosmeticItem(packet.userId, item.slot, item.asset_path);
                });
            }
        }

        // Apply high-speed positioning data matrices across the multiplayer floor
        avatar.getObjectByName("head_mesh").position.copy(packet.headPos);
        avatar.getObjectByName("left_hand_mesh").position.copy(packet.leftHandPos);
        avatar.getObjectByName("right_hand_mesh").position.copy(packet.rightHandPos);
    },

    /**
     * Storefront Interaction: Maps textures or geometry attachments to target slots
     */
    equipCosmeticItem(userId, slot, assetPath) {
        let avatar = (this.localAvatar && this.localAvatar.name === `avatar_${userId}`) 
            ? this.localAvatar 
            : this.remoteAvatars[userId];

        if (!avatar) return;

        const textureLoader = new THREE.TextureLoader();
        
        textureLoader.load(assetPath, (texture) => {
            if (slot === 'right_hand' || slot === 'left_hand') {
                const handMesh = avatar.getObjectByName(`${slot}_mesh`);
                if (handMesh) {
                    handMesh.material.map = texture;
                    handMesh.material.wireframe = false; // Turn off testing grid to reveal premium item look
                    handMesh.material.needsUpdate = true;
                    console.log(`Cosmetic applied successfully to avatar slot: ${slot}`);
                }
            }
        });
    }
};
```

```javascript
/**
```

```javascript
 * Phase 173 Production Lock
 * Feature: Hands-Only Velocity Card Mucking & Chip Splashing

AFRAME.registerComponent('hand-physics', {
  schema: {
    velocityThreshold: { type: 'number', default: 1.2 },
    targetMuckZone: { type: 'vec3', default: { x: 0, y: 0.8, z: -1.5 } }
  },

  init: function () {
    this.positionHistory = [];
    this.maxFrames = 5;
    this.trackedJoint = null;
    
    this.el.addEventListener('model-loaded', this.bindHandTracking.bind(this));
  },

  bindHandTracking: function () {
    const controllerMesh = this.el.getObject3D('mesh');
    if (controllerMesh) {
      this.trackedJoint = controllerMesh.getObject3D('wrist') || controllerMesh;
    }
  },

  tick: function (time, timeDelta) {
    if (!this.trackedJoint) return;

    const currentPos = new THREE.Vector3();
    this.trackedJoint.getWorldPosition(currentPos);

    this.positionHistory.push({ pos: currentPos.clone(), time: time });
    if (this.positionHistory.length > this.maxFrames) {
      this.positionHistory.shift();
    }

    if (this.positionHistory.length < 2) return;

    const oldest = this.positionHistory[0];
    const newest = this.positionHistory[this.positionHistory.length - 1];
    
    const distance = newest.pos.distanceTo(oldest.pos);
    const timeSec = (newest.time - oldest.time) / 1000;
    
    if (timeSec <= 0) return;
    const velocity = distance / timeSec;

    if (velocity > this.data.velocityThreshold) {
      this.evaluateFlickGesture(newest.pos, oldest.pos);
    }
  },

  evaluateFlickGesture: function (newestPos, oldestPos) {
    const direction = new THREE.Vector3().subVectors(newestPos, oldestPos).normalize();
    
    if (direction.z < -0.5) {
      this.positionHistory = [];
      this.el.emit('hand-physics-flick-triggered', { direction: direction });
      console.log("SVR Hand-Physics flick gesture processed successfully.");
    }
  }
});
```

```javascript
/**
 * Scarlett VR Poker - Daily Pick Wheel Module
 * Phase 173 Production Lock
 * Feature: $500 - $5000 Step Decay Rotation Engine
 */

AFRAME.registerComponent('daily-pick-wheel', {
  schema: {
    minReward: { type: 'number', default: 500 },
    maxReward: { type: 'number', default: 5000 },
    step: { type: 'number', default: 500 }
  },

  init: function () {
    this.isSpinning = false;
    this.el.addEventListener('click', this.startSpinEngine.bind(this));
  },

  startSpinEngine: function () {
    if (this.isSpinning) return;
    this.isSpinning = true;
    
    const totalSteps = ((this.data.maxReward - this.data.minReward) / this.data.step) + 1;
    const chosenIndex = Math.floor(Math.random() * totalSteps);
    const targetReward = this.data.minReward + (chosenIndex * this.data.step);

    const baseDegreesPerSegment = 360 / totalSteps;
    const extraRevolutions = 360 * (4 + Math.floor(Math.random() * 3));
    const targetDegrees = extraRevolutions + (chosenIndex * baseDegreesPerSegment);

    const startLocation = this.el.getAttribute('rotation') || { x: 0, y: 0, z: 0 };
    const initialZ = startLocation.z;

    let startTime = null;
    const duration = 5000;

    const animateWheel = (timestamp) => {
      if (!startTime) startTime = timestamp;
      const progress = timestamp - startTime;

      if (progress < duration) {
        const timeFactor = progress / duration;
        const decayFactor = 1 - Math.pow(1 - timeFactor, 3);
        const stepZ = initialZ + (targetDegrees * decayFactor);
        
        this.el.setAttribute('rotation', { x: startLocation.x, y: startLocation.y, z: stepZ });
        requestAnimationFrame(animateWheel);
      } else {
        const finalZ = (initialZ + targetDegrees) % 360;
        this.el.setAttribute('rotation', { x: startLocation.x, y: startLocation.y, z: finalZ });
        this.isSpinning = false;
        
        this.el.emit('daily-pick-finished', { rewardAmount: targetReward });
        console.log(`Daily Pick payout determined: $${targetReward}`);
      }
    };

    requestAnimationFrame(animateWheel);
  }
});
```

```javascript
    remoteHands.forEach((handEl) => {
```

```javascript
/**
 * Scarlett VR Poker - Velocity Hand Physics Module
 * Phase 173 Production Lock
 * Feature: Hands-Only Velocity Card Mucking & Chip Splashing
 */

AFRAME.registerComponent('hand-physics', {
  schema: {
    velocityThreshold: { type: 'number', default: 1.2 },
    targetMuckZone: { type: 'vec3', default: { x: 0, y: 0.8, z: -1.5 } }
  },

  init: function () {
    this.positionHistory = [];
    this.maxFrames = 5;
    this.trackedJoint = null;
    
    this.el.addEventListener('model-loaded', this.bindHandTracking.bind(this));
  },

  bindHandTracking: function () {
    const controllerMesh = this.el.getObject3D('mesh');
    if (controllerMesh) {
      this.trackedJoint = controllerMesh.getObject3D('wrist') || controllerMesh;
    }
  },

  tick: function (time, timeDelta) {
    if (!this.trackedJoint) return;

    const currentPos = new THREE.Vector3();
    this.trackedJoint.getWorldPosition(currentPos);

    this.positionHistory.push({ pos: currentPos.clone(), time: time });
    if (this.positionHistory.length > this.maxFrames) {
      this.positionHistory.shift();
    }

    if (this.positionHistory.length < 2) return;

    const oldest = this.positionHistory[0];
    const newest = this.positionHistory[this.positionHistory.length - 1];
    
    const distance = newest.pos.distanceTo(oldest.pos);
    const timeSec = (newest.time - oldest.time) / 1000;
    
    if (timeSec <= 0) return;
    const velocity = distance / timeSec;

    if (velocity > this.data.velocityThreshold) {
      this.evaluateFlickGesture(newest.pos, oldest.pos);
    }
  },

  evaluateFlickGesture: function (newestPos, oldestPos) {
    const direction = new THREE.Vector3().subVectors(newestPos, oldestPos).normalize();
    
    if (direction.z < -0.5) {
      this.positionHistory = [];
      this.el.emit('hand-physics-flick-triggered', { direction: direction });
      console.log("SVR Hand-Physics flick gesture processed successfully.");
    }
  }
});
```

```javascript
/**
 * Scarlett VR Poker - Daily Pick Wheel Module
 * Phase 173 Production Lock
 * Feature: $500 - $5000 Step Decay Rotation Engine
 */

AFRAME.registerComponent('daily-pick-wheel', {
  schema: {
    minReward: { type: 'number', default: 500 },
    maxReward: { type: 'number', default: 5000 },
    step: { type: 'number', default: 500 }
  },

  init: function () {
    this.isSpinning = false;
    this.el.addEventListener('click', this.startSpinEngine.bind(this));
  },

  startSpinEngine: function () {
    if (this.isSpinning) return;
    this.isSpinning = true;
    
    const totalSteps = ((this.data.maxReward - this.data.minReward) / this.data.step) + 1;
    const chosenIndex = Math.floor(Math.random() * totalSteps);
    const targetReward = this.data.minReward + (chosenIndex * this.data.step);

    const baseDegreesPerSegment = 360 / totalSteps;
    const extraRevolutions = 360 * (4 + Math.floor(Math.random() * 3));
    const targetDegrees = extraRevolutions + (chosenIndex * baseDegreesPerSegment);

    const startLocation = this.el.getAttribute('rotation') || { x: 0, y: 0, z: 0 };
    const initialZ = startLocation.z;

    let startTime = null;
    const duration = 5000;

    const animateWheel = (timestamp) => {
      if (!startTime) startTime = timestamp;
      const progress = timestamp - startTime;

      if (progress < duration) {
        const timeFactor = progress / duration;
        const decayFactor = 1 - Math.pow(1 - timeFactor, 3);
        const stepZ = initialZ + (targetDegrees * decayFactor);
        
        this.el.setAttribute('rotation', { x: startLocation.x, y: startLocation.y, z: stepZ });
        requestAnimationFrame(animateWheel);
      } else {
        const finalZ = (initialZ + targetDegrees) % 360;
        this.el.setAttribute('rotation', { x: startLocation.x, y: startLocation.y, z: finalZ });
        this.isSpinning = false;
        
        this.el.emit('daily-pick-finished', { rewardAmount: targetReward });
        console.log(`Daily Pick payout determined: $${targetReward}`);
      }
    };

    requestAnimationFrame(animateWheel);
  }
});
```

```javascript
const SVRNetworkModule = {
    socketInstance: null,
    broadcastInterval: null,
    transmitRateMS: 45, // Optimized update tickrate (~22 updates per second)

    init() {
        console.log("SVR Network Multiplexer Initialized.");
        this.establishServerConnection();
    },

    establishServerConnection() {
        // Automatically interfaces with your backend node routing layer
        this.socketInstance = new WebSocket(`wss://${window.location.host}/api/multiplayer/sync`);

        this.socketInstance.onopen = () => {
            console.log("Multiplayer network tunnel established successfully.");
            this.startReplicationLoop();
        };

        this.socketInstance.onmessage = (event) => {
            const incomingData = JSON.parse(event.data);
            this.demuxIncomingPacket(incomingData);
        };

        this.socketInstance.onclose = () => {
            console.warn("Network tunnel disconnected. Retrying connection...");
            clearInterval(this.broadcastInterval);
            setTimeout(() => this.establishServerConnection(), 5000);
        };
    },

    /**
     * Outbound Data Throttling: Captures tracked hand coordinates and ships them
     */
    startReplicationLoop() {
        this.broadcastInterval = setInterval(() => {
            if (!this.socketInstance || this.socketInstance.readyState !== WebSocket.OPEN) return;

            // Gather structural matrices directly from your camera and hand anchors
            const localPacket = {
                u_id: SVRPrivateModule.playerProfile?.user_id,
                h_rot: [world.camera.rotation.x, world.camera.rotation.y, world.camera.rotation.z],

                rh: [world.rightHandAnchor.position.x, world.rightHandAnchor.position.y, world.rightHandAnchor.position.z]
            };

            if (localPacket.u_id) {
                this.socketInstance.send(JSON.stringify(localPacket));
            }

        }));
```

To make sure the real-time movement looks incredibly fluid and natural when you are walking around and seeing other players, we can add **three critical network interpolation layers** to `js/scarlett1/mod_network.js`.

Without these updates, players might look like they are "teleporting" or jittering across the casino floor if someone has a brief lag spike or Wi-Fi drop.

Here is how we can upgrade the network script to handle smooth movement transitions perfectly:

---

## 1. Dead Reckoning & Linear Interpolation (Lerp)

Instead of immediately snapping an avatar head or hand mesh to the newest coordinates the second a data packet arrives, the avatar should smoothly slide from its old position to the new one over a split second.

* **The Improvement:** We can implement a `lerp` function inside the frame loop. If a packet drops, the avatar uses its last known velocity vector to "guess" where the player is stepping next, keeping their movement continuous until the next sync packet corrects it.

## 2. Dynamic Rotation Slerp (Spherical Interpolation)

When a player turns their head rapidly to look at the Chicago skyline or glances down at their wrist watch, sharp shifts in angles can look robotic.

* **The Improvement:** Use **Slerp** for head rotations. This calculates the shortest arc between two head orientation quaternions and eases the turn over a few milliseconds, making player head tilts look exactly like a real person nodding or looking around.

```javascript
                        // Safe conversion to quaternions for fluid rotation slerping
                        const targetQuaternion = new THREE.Quaternion().setFromEuler(targets.headRot);
                        headMesh.quaternion.slerp(targetQuaternion, 0.2);
                    }
                    if (leftHandMesh) leftHandMesh.position.lerp(targets.leftHandPos, 0.2);
                    if (rightHandMesh) rightHandMesh.position.lerp(targets.rightHandPos, 0.2);
                }
            }
        };
        requestAnimationFrame(renderLoop);
    }
};
```

```text
const SVRNetworkModule = {
    socketInstance: null, broadcastInterval: null, transmitRateMS: 45, targetPositions: {},
    init() { console.log("SVR Network Multiplexer Active."); this.establishServerConnection(); this.startInterpolationTick(); },
    establishServerConnection() {

        this.socketInstance.onopen = () => { console.log("Network tunnel tracking online."); this.startReplicationLoop(); };
        this.socketInstance.onmessage = (event) => { try { this.demuxIncomingPacket(JSON.parse(event.data)); } catch (err) { console.error("Bad tracking line packet parsing:", err); } };
        this.socketInstance.onclose = () => { clearInterval(this.broadcastInterval); setTimeout(() => this.establishServerConnection(), 5000); };
    },
    startReplicationLoop() {
        this.broadcastInterval = setInterval(() => {
            if (!this.socketInstance || this.socketInstance.readyState !== WebSocket.OPEN) return;
            const localPacket = {
                u_id: SVRPrivateModule?.playerProfile?.user_id || "guest_tester",
                h_rot: typeof world !== 'undefined' ? [world.camera.rotation.x, world.camera.rotation.y, world.camera.rotation.z] : [0, 0, 0],
                lh: typeof world !== 'undefined' && world.leftHandAnchor ? [world.leftHandAnchor.position.x, world.leftHandAnchor.position.y, world.leftHandAnchor.position.z] : [0, 0, 0],
                rh: typeof world !== 'undefined' && world.rightHandAnchor ? [world.rightHandAnchor.position.x, world.rightHandAnchor.position.y, world.rightHandAnchor.position.z] : [0, 0, 0]
            };
            this.socketInstance.send(JSON.stringify(localPacket));
        }, this.transmitRateMS);
    },
    demuxIncomingPacket(data) {
        this.targetPositions[data.u_id] = {
            headPos: new THREE.Vector3(data.lh[0], data.lh[1] + 0.5, data.lh[2]), leftHandPos: new THREE.Vector3(data.lh[0], data.lh[1], data.lh[2]), rightHandPos: new THREE.Vector3(data.rh[0], data.rh[1], data.rh[2]),
            headRot: new THREE.Euler(data.h_rot[0], data.h_rot[1], data.h_rot[2])
        };
    },
    startInterpolationTick() {
        const renderLoop = () => {
            requestAnimationFrame(renderLoop); if (typeof SVRAvatarModule === 'undefined' || !SVRAvatarModule.remoteAvatars) return;
            for (const userId in this.targetPositions) {
```

## 2. Security Frame Guard Module (`js/scarlett1/mod_security_bubble.js`)

```javascript
/**
 * SVR Poker - Spatial Anti-Cheat & Security Bubble Module
 * Filename: js/scarlett1/mod_security_bubble.js
 * Version: 5.5
 */

const SVRSecurityBubbleModule = {
    maxReachDistance: 1.8, // Maximum physical reach boundary in meters

    init() {
        console.log("SVR Spatial Security Bubble Activated.");
        window.addEventListener('svr_network_player_update', (event) => {
            this.validatePlayerBoundaries(event.detail);
        });
    },

    /**
     * Integrity Check: Ensures incoming player hand tracking vectors are physically possible
     */
    validatePlayerBoundaries(packet) {
        if (!packet.headPos || !packet.leftHandPos || !packet.rightHandPos) return;

        // Calculate direct distance between the user's head position and their tracking anchors
        const leftReach = packet.headPos.distanceTo(packet.leftHandPos);
        const rightReach = packet.headPos.distanceTo(packet.rightHandPos);

        // Flag anomalies if coordinates exceed natural physical movement limits
        if (leftReach > this.maxReachDistance || rightReach > this.maxReachDistance) {
            console.warn(`[SECURITY WARN] Spatial telemetry anomaly detected for User: ${packet.userId}`);
            // Enforces local interpolation dampening to isolate visual clipping
            packet.leftHandPos.clampLength(0, this.maxReachDistance);
            packet.rightHandPos.clampLength(0, this.maxReachDistance);
        }
    }
};
```

## 1. Node.js WebSocket Room Router (`backend/routes/multiplayer.js`)

This backend script dynamically groups players into their respective table threads (like the main poker pit, the Scorpion Room, or the Reiki room) and broadcasts positioning data with zero server-side storage overhead.

```javascript
/**
 * SVR Server - Real-Time Multiplayer Room Hub Router
 * Filename: backend/routes/multiplayer.js
 * Version: 5.5
 */

const WebSocket = require('ws');

// Track active connections by table room space
const activeTables = {}; // Format: { table_id: [ { userId, wsInstance } ] }

function initializeMultiplayerSync(server) {
    const wss = new WebSocket.Server({ noServer: true });

    // Handle initial secure upgrade handshake from client mod_network.js
    server.on('upgrade', (request, socket, head) => {
        if (request.url === '/api/multiplayer/sync') {
            wss.handleUpgrade(request, socket, head, (ws) => {
                wss.emit('connection', ws, request);
            });
        }
    });

    wss.on('connection', (ws) => {
        let currentTableId = null;
        let currentPlayerId = null;

        ws.on('message', (message) => {
            try {
                const packet = JSON.parse(message);
                currentPlayerId = packet.u_id;
                currentTableId = packet.t_id || "default_showroom_table";

                // Initialize table array if it doesn't exist yet
                if (!activeTables[currentTableId]) {
                    activeTables[currentTableId] = [];
                }

                // Register user connection tracking parameters if new to the table
                const playerSession = activeTables[currentTableId].find(p => p.userId === currentPlayerId);
                if (!playerSession) {
                    activeTables[currentTableId].push({ userId: currentPlayerId, wsInstance: ws });
                    console.log(`[NETWORK] Player ${currentPlayerId} joined sync thread: ${currentTableId}`);
                }

                // Broadcast spatial hand-tracking data to all OTHER players at this specific table
                activeTables[currentTableId].forEach(client => {
                    if (client.userId !== currentPlayerId && client.wsInstance.readyState === WebSocket.OPEN) {
                        client.wsInstance.send(JSON.stringify(packet));
                    }
                });

            } catch (err) {
                console.error("[NETWORK ERROR] Failed broadcasting serialization matrix:", err);
            }
        });

        // Clean up connection arrays immediately upon user disconnecting or leaving room boundary
        ws.on('close', () => {
            if (currentTableId && activeTables[currentTableId]) {
                activeTables[currentTableId] = activeTables[currentTableId].filter(client => client.userId !== currentPlayerId);
                console.log(`[NETWORK] Player ${currentPlayerId} disconnected from table thread.`);
                
                // Erase empty table arrays to conserve active system memory resources
                if (activeTables[currentTableId].length === 0) {
                    delete activeTables[currentTableId];
                }
            }
        });
    });

    console.log("SVR Backend Multiplayer Sync Pipeline Attached Sub-System Layer.");
}

module.exports = initializeMultiplayerSync;

```
